<!-- JavaScript for Aura Effect & Custom Cursor -->
<script>
  document.addEventListener('DOMContentLoaded', function() {
    const cursor = document.getElementById('custom-cursor');
    const cursorTrail = document.getElementById('cursor-trail');
    const auraContainer = document.getElementById('aura-container');
    let mouseX = 0, mouseY = 0;
    let trailX = 0, trailY = 0;
    
    document.addEventListener('mousemove', (e) => {
      mouseX = e.clientX;
      mouseY = e.clientY;
      
      cursor.style.left = mouseX - 10 + 'px';
      cursor.style.top = mouseY - 10 + 'px';
      
      createAura(mouseX, mouseY);
    });
    
    function animateTrail() {
      trailX += (mouseX - trailX) * 0.1;
      trailY += (mouseY - trailY) * 0.1;
      
      cursorTrail.style.left = trailX - 20 + 'px';
      cursorTrail.style.top = trailY - 20 + 'px';
      
      requestAnimationFrame(animateTrail);
    }
    animateTrail();
    
    function createAura(x, y) {
      const aura = document.createElement('div');
      aura.style.cssText = `
        position: absolute;
        left: ${x}px;
        top: ${y}px;
        width: 100px;
        height: 100px;
        background: radial-gradient(circle, rgba(0, 212, 255, 0.4) 0%, rgba(0, 100, 255, 0.2) 40%, transparent 70%);
        border-radius: 50%;
        transform: translate(-50%, -50%) scale(0);
        pointer-events: none;
        animation: auraExpand 0.8s ease-out forwards;
        filter: blur(2px);
      `;
      
      auraContainer.appendChild(aura);
      
      setTimeout(() => {
        aura.remove();
      }, 800);
    }
    
    const interactiveElements = document.querySelectorAll('a, button, .tech-card');
    interactiveElements.forEach(el => {
      el.addEventListener('mouseenter', () => {
        cursor.style.transform = 'scale(2)';
        cursor.style.borderColor = '#00ff88';
        cursor.style.boxShadow = '0 0 30px #00ff88, inset 0 0 15px rgba(0, 255, 136, 0.5)';
      });
      
      el.addEventListener('mouseleave', () => {
        cursor.style.transform = 'scale(1)';
        cursor.style.borderColor = '#00d4ff';
        cursor.style.boxShadow = '0 0 20px #00d4ff, inset 0 0 10px rgba(0, 212, 255, 0.5)';
      });
    });
    
    const style = document.createElement('style');
    style.textContent = `
      @keyframes auraExpand {
        0% {
          transform: translate(-50%, -50%) scale(0);
          opacity: 0.8;
        }
        100% {
          transform: translate(-50%, -50%) scale(3);
          opacity: 0;
        }
      }
    `;
    document.head.appendChild(style);
  });
</script>
